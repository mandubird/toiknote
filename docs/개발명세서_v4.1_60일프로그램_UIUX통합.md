### 계산 로직 (1차 버전 - Rule Based)

```javascript
function calculateEstimatedScore(userData) {
  const { partAccuracy, part7AvgTime } = userData
  
  // 1. LC 점수 계산
  const LC_accuracy = (
    partAccuracy[1] + 
    partAccuracy[2] + 
    partAccuracy[3] + 
    partAccuracy[4]
  ) / 4
  
  let LC_score = LC_accuracy * 495  // LC 만점 495
  
  // 2. RC 점수 계산
  const RC_accuracy = (
    partAccuracy[5] * 1.1 +  // Part 5 가중치
    partAccuracy[6] * 1.0 +
    partAccuracy[7] * 1.3    // Part 7 가중치 (가장 높음)
  ) / 3.4
  
  let RC_score = RC_accuracy * 495  // RC 만점 495
  
  // 3. Part 7 시간 패널티
  const benchmarkTime = 50  // 분
  let timePenalty = 0
  
  if (part7AvgTime > benchmarkTime) {
    const excessMinutes = part7AvgTime - benchmarkTime
    
    if (excessMinutes <= 3) {
      timePenalty = -10
    } else if (excessMinutes <= 7) {
      timePenalty = -20
    } else {
      timePenalty = -30
    }
  }
  
  // 4. 최종 점수
  const estimatedScore = Math.round(LC_score + RC_score + timePenalty)
  
  return {
    estimatedScore: estimatedScore,
    LC_score: Math.round(LC_score),
    RC_score: Math.round(RC_score),
    timePenalty: timePenalty
  }
}
```

### 약점 TOP 3 계산

```javascript
async function calculateWeakTOP3(userId) {
  // 1. 태그별 오답률 조회
  const { data: tagStats } = await supabase
    .from('tag_stats')
    .select('*')
    .eq('user_id', userId)
    .single()
  
  // 2. 최근 2주 개선 여부 확인
  const twoWeeksAgo = new Date()
  twoWeeksAgo.setDate(twoWeeksAgo.getDate() - 14)
  
  const { data: recentWrong } = await supabase
    .from('wrong_notes')
    .select('auto_tags')
    .eq('user_id', userId)
    .gte('created_at', twoWeeksAgo.toISOString())
  
  const recentTagCounts = {}
  recentWrong.forEach(note => {
    note.auto_tags.forEach(tag => {
      if (!recentTagCounts[tag]) recentTagCounts[tag] = 0
      recentTagCounts[tag]++
    })
  })
  
  // 3. 오답률 높은 순 + 최근 개선 없음 정렬
  const weaknesses = Object.entries(tagStats.tag_counts)
    .map(([tag, count]) => ({
      tag: tag,
      totalWrong: count,
      recentWrong: recentTagCounts[tag] || 0,
      wrongRate: (count / tagStats.total_wrong * 100).toFixed(1),
      noImprovement: recentTagCounts[tag] >= count * 0.3  // 최근 2주에도 30% 이상
    }))
    .sort((a, b) => {
      // 1순위: 개선 없음
      if (a.noImprovement && !b.noImprovement) return -1
      if (!a.noImprovement && b.noImprovement) return 1
      
      // 2순위: 오답률
      return b.wrongRate - a.wrongRate
    })
    .slice(0, 3)
  
  return weaknesses
}
```

        .from('users')
        .update({
          current_week: calculatedWeek,
          last_week_update: now
        })

  }
  
  next()
}
