```tsx
// src/components/landing/ScarcityCounter.tsx (신규)
// 위치: 가격 섹션 바로 위

// DB 조회 함수 (신규 — 기존 함수 수정 금지)
// src/services/landingService.ts 에 추가

export async function getRemainingSlots(): Promise<number> {
  const BETA_LIMIT = 100

  const { count } = await supabase
    .from('subscriptions')
    .select('*', { count: 'exact', head: true })
    .eq('plan_name', '8week_beta')  // pricing_plans 기준

  const used = count ?? 0
  return Math.max(0, BETA_LIMIT - used)
}

// 컴포넌트
export function ScarcityCounter() {
  const [remaining, setRemaining] = useState<number | null>(null)

  useEffect(() => {
    getRemainingSlots().then(setRemaining)
  }, [])

  if (remaining === null) return null
  if (remaining <= 0) return (
    <div className="scarcity-sold-out">
      ❌ 베타 마감 — 다음 기수 대기 신청
    </div>
  )

  // 남은 자리 30개 이하일 때만 표시 (너무 많으면 희소성 효과 없음)
  if (remaining > 30) return null

  return (
    <div className={`scarcity-counter ${remaining <= 10 ? 'urgent' : 'warning'}`}>
      <span className="icon">🔥</span>
      <strong>현재 베타 {remaining}자리 남음</strong>
      <span className="sub">베타 100명 한정 · 이후 정가 적용</span>
    </div>
  )
}
```

### 2-4. 트리거 4 — 사회적 증명 (실시간 DB 연동)

```tsx
// src/components/landing/SocialProofBar.tsx (신규)
// 위치: Hero Section 신뢰 배지 영역 교체

// DB 집계 함수 (신규 — 기존 통계 함수 수정 금지)
// src/services/landingService.ts 에 추가

export async function getSocialProofStats(): Promise<{
  monthlySuccessCount: number  // 이번 달 900 돌파 수
  avgScoreGain: number         // 전체 평균 점수 상승
}> {
  const monthStart = new Date(
    new Date().getFullYear(), new Date().getMonth(), 1
  ).toISOString()

  // 이번 달 900 돌파 (badge_level = '900' + 이번 달 갱신)
  const { count: monthlySuccess } = await supabase
    .from('users')
    .select('*', { count: 'exact', head: true })
    .eq('badge_level', '900')
    .gte('updated_at', monthStart)

  // 전체 후기 기반 평균 점수 상승
  const { data: reviews } = await supabase
    .from('reviews')
    .select('score_before, score_after')
    .eq('approved', true)
    .not('score_before', 'is', null)
    .not('score_after', 'is', null)

  const avgGain = reviews && reviews.length > 0
    ? Math.round(
        reviews.reduce((sum, r) => sum + (r.score_after - r.score_before), 0) / reviews.length
      )
    : 62  // fallback (초기 데이터 없을 때 기본값)

  return {
    monthlySuccessCount: monthlySuccess ?? 17,  // fallback
    avgScoreGain: avgGain
  }
}

// 컴포넌트
export function SocialProofBar() {
  const [stats, setStats] = useState({ monthlySuccessCount: 17, avgScoreGain: 62 })

  useEffect(() => {
    getSocialProofStats().then(setStats)
  }, [])

  return (
    <div className="social-proof-bar">
      <div className="proof-item">
        <strong>{stats.monthlySuccessCount}명</strong>
        <span>이번 달 900 돌파</span>
      </div>
      <div className="divider" />
      <div className="proof-item">
        <strong>+{stats.avgScoreGain}점</strong>
        <span>평균 점수 상승</span>
      </div>
      <div className="divider" />
      <div className="proof-item">
        <strong>60일</strong>
        <span>집중 관리</span>
      </div>
    </div>
  )
}
```

### 2-5. 트리거 5 — 즉시 행동 유도 (CTA 3개 통일)

```tsx
// ⚠️ 전체 페이지에서 CTA 버튼은 아래 3개만 사용
// 다른 버튼 텍스트 사용 금지

// CTA 상수 정의 — src/constants/cta.ts (신규)
export const CTA = {
  FREE_DIAGNOSIS: '무료 진단 시작',         // 버튼 1: 미결제 사용자용
  VIEW_RESULT:    '결과 보기',              // 버튼 2: 진단 완료 사용자용
  START_PROGRAM:  '900 돌파 프로그램 시작'  // 버튼 3: 결제 유도
} as const

// CTA 버튼 컴포넌트 — src/components/landing/CtaButton.tsx (신규)
import { CTA } from '@/constants/cta'
import { useNavigate } from 'react-router-dom'

type CtaType = keyof typeof CTA

interface CtaButtonProps {
  type: CtaType
  position: string   // GA 트래킹용 위치 라벨
  size?: 'normal' | 'large'
}

export function CtaButton({ type, position, size = 'normal' }: CtaButtonProps) {
  const navigate = useNavigate()

  const handleClick = () => {
    // GA 이벤트 트래킹
    if (typeof window.gtag === 'function') {
      window.gtag('event', 'cta_click', {
        event_category: 'conversion',
        event_label: `${type}_${position}`
      })
    }
    // Meta Pixel
    if (typeof fbq === 'function') {
      if (type === 'START_PROGRAM') fbq('track', 'InitiateCheckout')
      if (type === 'FREE_DIAGNOSIS') fbq('track', 'Lead')
    }

    // 라우팅
    if (type === 'FREE_DIAGNOSIS') navigate('/diagnostic')
    if (type === 'VIEW_RESULT')    navigate('/dashboard')
    if (type === 'START_PROGRAM')  navigate('/upgrade')
  }

  return (
    <button
      className={`cta-button cta-${type.toLowerCase()} ${size}`}
      onClick={handleClick}
    >
      {CTA[type]}
    </button>
  )
}
```
