## 🧩 2. 컴포넌트 구현

### 2-1. ImageSourceBottomSheet (신규 컴포넌트)

```tsx
// src/components/ImageSourceBottomSheet.tsx (신규 파일)

import { useRef } from 'react'

interface ImageSourceBottomSheetProps {
  isOpen: boolean
  onClose: () => void
  onImagesSelected: (files: File[]) => void  // 카메라/갤러리 공통 콜백
}

export function ImageSourceBottomSheet({
  isOpen,
  onClose,
  onImagesSelected
}: ImageSourceBottomSheetProps) {
  const cameraInputRef = useRef<HTMLInputElement>(null)
  const galleryInputRef = useRef<HTMLInputElement>(null)

  const handleFiles = (e: React.ChangeEvent<HTMLInputElement>) => {
    const files = Array.from(e.target.files ?? [])
    if (files.length === 0) return

    // 파일 유효성 검사
    const validFiles = files.filter(file => {
      const isValidType = ['image/jpeg', 'image/png', 'image/heic', 'image/webp']
        .includes(file.type)
      const isValidSize = file.size <= 10 * 1024 * 1024  // 10MB
      return isValidType && isValidSize
    })

    if (validFiles.length === 0) {
      alert('jpg, png, heic 형식의 10MB 이하 이미지만 가능합니다.')
      return
    }

    onImagesSelected(validFiles)
    onClose()

    // input 초기화 (같은 파일 재선택 가능하게)
    e.target.value = ''
  }

  if (!isOpen) return null

  return (
    <>
      {/* 딤 배경 */}
      <div
        className="bottomsheet-overlay"
        onClick={onClose}
      />

      {/* 바텀시트 */}
      <div className="bottomsheet">
        <div className="bottomsheet-handle" />

        <p className="bottomsheet-title">문제 사진 추가</p>

        {/* 카메라 촬영 */}
        <button
          className="bottomsheet-option"
          onClick={() => cameraInputRef.current?.click()}
        >
          <span className="option-icon">📷</span>
          <span className="option-label">카메라로 촬영</span>
        </button>

        {/* 갤러리에서 선택 */}
        <button
          className="bottomsheet-option"
          onClick={() => galleryInputRef.current?.click()}
        >
          <span className="option-icon">🖼️</span>
          <span className="option-label">갤러리에서 선택</span>
          <span className="option-sub">최대 3장</span>
        </button>

        <button className="bottomsheet-cancel" onClick={onClose}>
          취소
        </button>
      </div>

      {/* 카메라 input — capture="environment" (후면 카메라) */}
      <input
        ref={cameraInputRef}
        type="file"
        accept="image/*"
        capture="environment"
        style={{ display: 'none' }}
        onChange={handleFiles}
      />

      {/* 갤러리 input — capture 없음, multiple 허용 */}
      <input
        ref={galleryInputRef}
        type="file"
        accept="image/jpeg,image/png,image/heic,image/webp"
        multiple
        style={{ display: 'none' }}
        onChange={(e) => {
          const files = Array.from(e.target.files ?? [])
          // 최대 3장 제한
          if (files.length > 3) {
            alert('최대 3장까지 선택 가능합니다.')
            e.target.value = ''
            return
          }
          handleFiles(e)
        }}
      />
    </>
  )
}
```

## 📋 4. 플랫폼별 동작 정리

| 환경 | 카메라 촬영 | 갤러리 선택 |
|------|-----------|-----------|
| iOS Safari | ✅ 후면 카메라 바로 실행 | ✅ 사진 보관함 |
| Android Chrome | ✅ 카메라 앱 실행 | ✅ 파일 선택기 |
| 맥북/PC | ✅ 웹캠 (있는 경우) | ✅ 파일 탐색기 |
| PWA 설치 후 | ✅ 네이티브 수준 | ✅ 네이티브 수준 |

**핵심 차이:**
```
카메라: <input type="file" accept="image/*" capture="environment">
갤러리: <input type="file" accept="image/*" multiple>
        (capture 속성 없음 — 이게 핵심)
```
