##### 1-3. 가중치 적용 함수
```javascript
async function calculateWeightedTagPenalty(userId) {
  // 1. 사용자 점수 구간 조회
  const user = await supabase
    .from('users')
    .select('current_score')
    .eq('id', userId)
    .single()
  
  const scoreRange = getScoreRange(user.data.current_score)  // "700-800"
  
  // 2. 사용자 태그별 오답 조회
  const { data: wrongNotes } = await supabase
    .from('wrong_notes')
    .select('part, auto_tags, difficulty')
    .eq('user_id', userId)
  
  // 3. 태그 가중치 조회
  const { data: weights } = await supabase
    .from('tag_weights')
    .select('*')
  
  const weightMap = {}
  weights.forEach(w => {
    const key = `${w.part}-${w.tag_name}`
    weightMap[key] = {
      base: w.score_impact_weight,
      difficulty: w.difficulty_weight,
      scoreRange: w.score_range_weight
    }
  })
  
  // 4. 가중 패널티 계산
  let totalPenalty = 0
  const tagCount = {}
  
  wrongNotes.forEach(note => {
    note.auto_tags.forEach(tag => {
      const key = `${note.part}-${tag}`
      const weight = weightMap[key]
      
      if (!weight) return
      
      // 기본 가중치 × 난이도 가중치 × 점수구간 가중치
      const finalWeight = 
        weight.base * 
        (weight.difficulty[note.difficulty.toString()] || 1.0) *
        (weight.scoreRange[scoreRange] || 1.0)
      
      totalPenalty += finalWeight * 5
      
      if (!tagCount[tag]) tagCount[tag] = 0
      tagCount[tag]++
    })
  })
  
  return {
    totalPenalty: Math.round(totalPenalty),
    weightedTags: tagCount
  }
}
```

##### 2-2. accuracy_stats 뷰 생성 (자동 계산)
```sql
CREATE VIEW accuracy_stats AS
SELECT 
  user_id,
  part,
  
  -- 전체 정답률
  COUNT(*) as total_solved,
  COUNT(*) FILTER (WHERE is_correct = TRUE) as correct_count,
  COUNT(*) FILTER (WHERE is_correct = FALSE) as wrong_count,
  ROUND(
    COUNT(*) FILTER (WHERE is_correct = TRUE)::numeric / 
    NULLIF(COUNT(*), 0) * 100, 
    1
  ) as accuracy_rate,
  
  -- 최근 30일 정답률
  ROUND(
    COUNT(*) FILTER (WHERE is_correct = TRUE AND attempt_date >= NOW() - INTERVAL '30 days')::numeric / 
    NULLIF(COUNT(*) FILTER (WHERE attempt_date >= NOW() - INTERVAL '30 days'), 0) * 100,
    1
  ) as accuracy_rate_30d,
  
  -- 평균 풀이 시간
  AVG(solving_time) as avg_solving_time,
  AVG(solving_time) FILTER (WHERE is_correct = TRUE) as avg_time_correct,
  AVG(solving_time) FILTER (WHERE is_correct = FALSE) as avg_time_wrong

FROM solved_questions
GROUP BY user_id, part;
```

```javascript
async function getUserAccuracyStats(userId) {
  const { data } = await supabase
    .from('accuracy_stats')
    .select('*')
    .eq('user_id', userId)
  
  // 파트별 정답률 객체 생성
  const accuracyByPart = {}

    accuracyByPart[stat.part] = {
      totalSolved: stat.total_solved,
      correctCount: stat.correct_count,
      wrongCount: stat.wrong_count,
      accuracyRate: stat.accuracy_rate,

  // score_analytics에 저장
  await supabase
    .from('score_analytics')
    .upsert({
      user_id: userId,
      part_accuracy: accuracyByPart
    })
  
  return accuracyByPart
}
```

##### 3-2. 사용자 약점과 구간 패턴 매칭 함수
```javascript
async function diagnoseUserWeakness(userId) {
  // 1. 사용자 점수 조회
  const { data: user } = await supabase
    .from('users')
    .select('current_score, estimated_score')
    .eq('id', userId)
    .single()
  
  const scoreRange = getScoreRange(user.estimated_score || user.current_score)
  
  // 2. 해당 구간의 일반적 패턴 조회
  const { data: pattern } = await supabase
    .from('score_band_patterns')
    .select('*')
    .eq('score_range', scoreRange)
    .single()
  
  // 3. 사용자 실제 약점 조회
  const { data: tagStats } = await supabase
    .from('tag_stats')
    .select('*')
    .eq('user_id', userId)
    .single()
  
  const { data: accuracyStats } = await supabase
    .from('accuracy_stats')
    .select('*')
    .eq('user_id', userId)
  
  // 4. 패턴 매칭 + 개인화
  const diagnosis = {
    scoreRange: scoreRange,
    bandPattern: pattern.common_weaknesses,
    
    // 사용자 실제 약점
    actualWeakParts: findWeakParts(accuracyStats),
    actualWeakTags: findWeakTags(tagStats),
    
    // 매칭 결과
    matchedWeakness: matchPattern(pattern, tagStats, accuracyStats),
    
    // 권장 전략
    recommendedStrategy: pattern.recommended_strategy,
    expectedImprovement: pattern.avg_improvement_60days
  }
  
  return diagnosis
}

function matchPattern(pattern, tagStats, accuracyStats) {
  // 구간 패턴과 사용자 실제 약점 비교
  const matches = []
  
  Object.entries(pattern.common_weaknesses.parts).forEach(([part, info]) => {
    const userAccuracy = accuracyStats.find(a => a.part === parseInt(part))
    
    if (userAccuracy && userAccuracy.accuracy_rate < 70) {
      matches.push({
        part: part,
        patternWeakness: info.weakness,
        userAccuracy: userAccuracy.accuracy_rate,
        priority: info.priority,
        confirmed: true
      })
    }
  })
  
  return matches
}
```

```javascript
async function performCompleteDiagnosis(userId) {
  // 1. 점수 구간 판정
  const { data: user } = await supabase
    .from('users')
    .select('*')
    .eq('id', userId)
    .single()
  
  const scoreRange = getScoreRange(user.estimated_score || user.current_score)
  
  // 2. 구간별 약점 패턴 조회
  const bandDiagnosis = await diagnoseUserWeakness(userId)
  
  // 3. 파트별 정답률 조회
  const accuracyStats = await getUserAccuracyStats(userId)
  
  // 4. 태그별 가중 패널티 계산
  const weightedPenalty = await calculateWeightedTagPenalty(userId)
  
  // 5. 종합 진단 생성
  const completeDiagnosis = {
    // 기본 정보
    userId: userId,
    currentScore: user.current_score,
    estimatedScore: user.estimated_score,
    targetScore: user.target_score,
    
    // 구간 정보
    scoreRange: scoreRange,
    scoreBandPattern: bandDiagnosis.bandPattern,
    
    // 약점 분석
    weakParts: bandDiagnosis.actualWeakParts,
    weakTags: bandDiagnosis.actualWeakTags,
    weightedTagPenalty: weightedPenalty,
    
    // 정답률
    partAccuracy: accuracyStats,
    
    // 진단 결과
    primaryWeakness: bandDiagnosis.bandPattern.primary,
    secondaryWeaknesses: bandDiagnosis.bandPattern.secondary,
    confirmedWeaknesses: bandDiagnosis.matchedWeakness,
    
    // 권장 사항
    recommendedStrategy: bandDiagnosis.recommendedStrategy,
    expectedImprovement: bandDiagnosis.expectedImprovement,
    
    diagnosedAt: new Date()
  }
  
  // 6. score_analytics에 저장
  await supabase
    .from('score_analytics')
    .upsert({
      user_id: userId,
      ...completeDiagnosis,
      last_analyzed_at: new Date()
    })
  
  return completeDiagnosis
}
```

```javascript
async function generatePersonalizedProgram(userId) {
  // 1. 종합 진단 수행
  const diagnosis = await performCompleteDiagnosis(userId)
  
  // 2. 약점 우선순위 정렬
  const prioritizedWeaknesses = diagnosis.confirmedWeaknesses
    .sort((a, b) => a.priority - b.priority)
  
  // 3. 8주 루틴 생성
  const program = []
  
  // Week 1-2: 1순위 약점 집중
  const week1_2 = prioritizedWeaknesses[0]
  program.push({
    week: 1,
    focusParts: [parseInt(week1_2.part)],
    focusTags: getTagsForWeakness(week1_2.patternWeakness),
    dailyTaskCount: 25,
    strategyText: `${week1_2.patternWeakness} 집중 공략 (1순위 약점)`,
    customized: true
  })
  
  program.push({
    week: 2,
    focusParts: [parseInt(week1_2.part)],
    focusTags: getTagsForWeakness(week1_2.patternWeakness),
    dailyTaskCount: 30,
    strategyText: `${week1_2.patternWeakness} 완성 주차`,
    customized: true
  })
  
  // Week 3-4: 2순위 약점
  if (prioritizedWeaknesses[1]) {
    const week3_4 = prioritizedWeaknesses[1]
    program.push({
      week: 3,
      focusParts: [parseInt(week3_4.part)],
      focusTags: getTagsForWeakness(week3_4.patternWeakness),
      dailyTaskCount: 25,
      strategyText: `${week3_4.patternWeakness} 집중 (2순위)`,
      customized: true
    })
    
    program.push({
      week: 4,
      focusParts: [parseInt(week3_4.part)],
      focusTags: getTagsForWeakness(week3_4.patternWeakness),
      dailyTaskCount: 30,
      strategyText: `${week3_4.patternWeakness} 완성`,
      customized: true
    })
  }
  
  // Week 5-6: LC 보완 or 3순위 약점
```

```javascript
async function generateWeeklyReport(userId, completedWeek) {
  // 1. 주차 시작/종료 스냅샷 조회
  const { data: snapshots } = await supabase
    .from('weekly_snapshots')
    .select('*')
    .eq('user_id', userId)
    .eq('week', completedWeek)
    .order('snapshot_date')
  
  const startSnap = snapshots.find(s => s.snapshot_type === 'week_start')
  const endSnap = snapshots.find(s => s.snapshot_type === 'week_end')
  
  if (!startSnap || !endSnap) {
    throw new Error('스냅샷 데이터 부족')
  }
  
  // 2. 오답 감소율 계산
  const wrongReduction = {
    startWrong: startSnap.total_wrong,
    endWrong: endSnap.total_wrong,
    reductionCount: startSnap.total_wrong - endSnap.total_wrong,
    reductionRate: ((startSnap.total_wrong - endSnap.total_wrong) / startSnap.total_wrong * 100).toFixed(1)
  }
  
  // 3. 정확도 변화
  const accuracyChange = {
    start: startSnap.accuracy_rate,
    end: endSnap.accuracy_rate,
    change: (endSnap.accuracy_rate - startSnap.accuracy_rate).toFixed(1)
  }
  
  // 4. Part 7 시간 변화
  const part7TimeChange = {
    start: startSnap.part_avg_time['7'],
    end: endSnap.part_avg_time['7'],
    change: endSnap.part_avg_time['7'] - startSnap.part_avg_time['7']
  }
  
  // 5. 태그별 개선도
  const tagImprovement = {}
  Object.keys(startSnap.tag_wrong_counts).forEach(tag => {
    const startCount = startSnap.tag_wrong_counts[tag]
    const endCount = endSnap.tag_wrong_counts[tag] || 0
    tagImprovement[tag] = {
      start: startCount,
      end: endCount,
      reduction: startCount - endCount,
      reductionRate: ((startCount - endCount) / startCount * 100).toFixed(1)
    }
  })
  
  // 6. 점수 변화
  const scoreChange = {
    start: startSnap.estimated_score,
    end: endSnap.estimated_score,
    improvement: endSnap.estimated_score - startSnap.estimated_score
  }
  
  // 7. GPT로 다음 주 전략 생성
  const nextWeekStrategy = await generateNextWeekStrategy(userId, completedWeek + 1, {
    wrongReduction,
    accuracyChange,
    part7TimeChange,
    scoreChange
  })
  
  // 8. weekly_reports에 저장
  await supabase
    .from('weekly_reports')
    .insert({
      user_id: userId,
      week: completedWeek,
      
      // 오답 감소율 🆕
      wrong_reduction_rate: parseFloat(wrongReduction.reductionRate),
      wrong_reduction_count: wrongReduction.reductionCount,
      
      // 정확도 변화
      accuracy_change: parseFloat(accuracyChange.change),
      
      // Part 7 시간 변화 🆕
      part7_time_change: part7TimeChange.change,
      
      // 태그별 개선도 🆕
      weak_tags_improvement: tagImprovement,
      
      // 점수 변화
      estimated_score_start: scoreChange.start,
      estimated_score_end: scoreChange.end,
      score_improvement: scoreChange.improvement,
      
      // 다음 주 전략
      next_week_strategy: nextWeekStrategy
    })
  
  return {
    wrongReduction,
    accuracyChange,
    part7TimeChange,
    tagImprovement,
    scoreChange,
    nextWeekStrategy
  }
}
```

```javascript
import puppeteer from 'puppeteer'
import fs from 'fs'
import Handlebars from 'handlebars'

async function generateWeeklyReportPDF(userId, week) {
  // 1. 리포트 데이터 조회
  const { data: report } = await supabase
    .from('weekly_reports')
    .select('*')
    .eq('user_id', userId)
    .eq('week', week)
    .single()
  
  const { data: user } = await supabase
    .from('users')
    .select('display_name, email')
    .eq('id', userId)
    .single()
  
  // 2. HTML 템플릿 로드
  const templateHtml = fs.readFileSync('./templates/weekly-report.html', 'utf8')
  const template = Handlebars.compile(templateHtml)
  
  // 3. 데이터 매핑
  const html = template({
    week: week,
    userName: user.display_name,
    dateRange: getWeekDateRange(week),
    wrongReductionRate: report.wrong_reduction_rate,
    accuracyChange: report.accuracy_change,
    accuracyChangeClass: report.accuracy_change > 0 ? 'improvement' : 'decline',
    part7TimeChange: Math.abs(report.part7_time_change),
    scoreStart: report.estimated_score_start,
    scoreEnd: report.estimated_score_end,
    scoreImprovement: report.score_improvement,
    tagLabels: JSON.stringify(Object.keys(report.weak_tags_improvement)),
    tagData: JSON.stringify(Object.values(report.weak_tags_improvement).map(t => t.end)),
    nextWeekStrategy: report.next_week_strategy,
    generatedAt: new Date().toLocaleDateString('ko-KR')
  })
  
  // 4. Puppeteer로 PDF 생성
  const browser = await puppeteer.launch()
  const page = await browser.newPage()
  await page.setContent(html, { waitUntil: 'networkidle0' })
  
  const pdfBuffer = await page.pdf({
    format: 'A4',
    printBackground: true,
    margin: {
      top: '20px',
      bottom: '20px',
      left: '20px',
      right: '20px'
    }
  })
  
  await browser.close()
  
  // 5. Supabase Storage에 업로드
  const fileName = `reports/${userId}/week-${week}-${Date.now()}.pdf`
  const { data: uploadData, error } = await supabase.storage
    .from('weekly-reports')
    .upload(fileName, pdfBuffer, {
      contentType: 'application/pdf',
      cacheControl: '3600'
    })
  
  if (error) throw error
  
  // 6. 공개 URL 생성
  const { data: publicUrl } = supabase.storage
    .from('weekly-reports')
    .getPublicUrl(fileName)
  
  // 7. weekly_reports에 PDF URL 저장
  await supabase
    .from('weekly_reports')
    .update({ pdf_url: publicUrl.publicUrl })
    .eq('user_id', userId)
    .eq('week', week)
  
  return publicUrl.publicUrl
}
```
