## ⚠️ 충돌 경고 (커서 AI 작업 전 필독)

### 경고 1 — tag_weights 테이블 폐기
- `v4.2`에서 생성한 `tag_weights (part, tag_name, weight)` 테이블은 **`tag_master.weight`로 흡수**
- 기존 `tag_weights` 데이터는 마이그레이션 후 삭제

### 경고 2 — wrong_answers.auto_tags 유지
- `wrong_answers.auto_tags TEXT[]` 컬럼은 **레거시 호환용으로 유지**
- 신규 로직은 `question_tags` 테이블 사용
- 기존 `auto_tags` 기반 집계 코드는 `user_tag_stats` 기반으로 단계적 교체

### 경고 3 — user_program_plans.focus_tags 타입 변경
- 기존: `focus_tags TEXT[]`
- 변경: `focus_tag_ids UUID[]` 컬럼 추가 (기존 컬럼 유지, 신규 컬럼 병행)

## 🧮 2. 약점 점수 계산 로직 (weakness_score)

### 2-1. 공식 확정

```
weakness_score =
  (1 - accuracy_rate) × weight
  + time_penalty
  + (depth × 0.05)
```

**time_penalty 기준:**
```
기준시간 = 태그별 기본값 (P7: 90초, RC: 60초, LC: 30초)

average_time ≤ 기준시간           → penalty = 0
기준시간 < average_time ≤ 기준×1.2 → penalty = 0.1
average_time > 기준시간×1.2       → penalty = 0.2
```

**안정화 조건:**
```
attempt_count < 3 → 랭킹 제외 (데이터 부족)
최근 30일 내 시도 있는 태그 → weight × 1.2 추가 적용
```

### 2-2. weakness_score 갱신 함수

```typescript
// src/services/tagStatsService.ts (신규 파일)

import { supabase } from '@/lib/supabase'

// 기준 시간 (초)
const TIME_BENCHMARKS: Record<string, number> = {
  'P7': 90,
  'P5': 60,
  'P6': 60,
  'RC': 60,
  'LC': 30,
  'META': 0  // 시간 패널티 없음
}

export async function updateUserTagStats(
  userId: string,
  tagId: string,
  isCorrect: boolean,
  solveTimeSeconds: number
): Promise<void> {
  // 1. 태그 메타 조회
  const { data: tag } = await supabase
    .from('tag_master')
    .select('weight, depth, part, category')
    .eq('id', tagId)
    .single()

  if (!tag) return

  // 2. 기존 stats 조회 (없으면 신규 생성)
  const { data: existing } = await supabase
    .from('user_tag_stats')
    .select('*')
    .eq('user_id', userId)
    .eq('tag_id', tagId)
    .single()

  const prevAttempt = existing?.attempt_count ?? 0
  const prevCorrect = existing?.correct_count ?? 0
  const prevAvgTime = existing?.average_time_seconds ?? 0

  const newAttempt = prevAttempt + 1
  const newCorrect = prevCorrect + (isCorrect ? 1 : 0)
  const newAvgTime = ((prevAvgTime * prevAttempt) + solveTimeSeconds) / newAttempt
  const accuracyRate = newCorrect / newAttempt

  // 3. 시간 패널티 계산
  const benchmark = TIME_BENCHMARKS[tag.part ?? tag.category] ?? 60
  let timePenalty = 0
  if (newAvgTime > benchmark * 1.2) timePenalty = 0.2
  else if (newAvgTime > benchmark) timePenalty = 0.1

  // 4. 최근 30일 가중치
  const recentWeight = existing?.last_attempted_at &&
    new Date(existing.last_attempted_at) > new Date(Date.now() - 30 * 24 * 60 * 60 * 1000)
    ? tag.weight * 1.2
    : tag.weight

  // 5. weakness_score 계산
  const weaknessScore = (1 - accuracyRate) * recentWeight
    + timePenalty
    + (tag.depth * 0.05)

  // 6. upsert
  await supabase
    .from('user_tag_stats')
    .upsert({
      user_id: userId,
      tag_id: tagId,
      attempt_count: newAttempt,
      correct_count: newCorrect,
      average_time_seconds: newAvgTime,
      weakness_score: weaknessScore,
      last_attempted_at: new Date(),
      updated_at: new Date()
    }, { onConflict: 'user_id, tag_id' })
}
```

### 2-3. Top10 약점 랭킹 조회

```typescript
export interface WeakTag {
  tagId: string
  tagCode: string
  tagName: string
  category: string
  part: string
  weaknessScore: number
  accuracyRate: number
  attemptCount: number
  averageTime: number
}

export async function getTop10WeakTags(userId: string): Promise<WeakTag[]> {
  const { data } = await supabase
    .from('user_tag_stats')
    .select(`
      weakness_score,
      accuracy_rate,
      attempt_count,
      average_time_seconds,
      tag:tag_id (
        id, tag_code, tag_name, category, part
      )
    `)
    .eq('user_id', userId)
    .gte('attempt_count', 3)          // 안정화 조건
    .order('weakness_score', { ascending: false })
    .limit(10)

  return (data ?? []).map(row => ({
    tagId: row.tag.id,
    tagCode: row.tag.tag_code,
    tagName: row.tag.tag_name,
    category: row.tag.category,
    part: row.tag.part,
    weaknessScore: row.weakness_score,
    accuracyRate: row.accuracy_rate,
    attemptCount: row.attempt_count,
    averageTime: row.average_time_seconds
  }))
}
```

### 3-2. 1차 룰 기반 태깅

```typescript
// src/services/autoTagService.ts (신규 파일)

// 키워드 → tag_code 매핑 룰
const KEYWORD_RULES: Record<string, string[]> = {
  'G-001': ['subject-verb', '수일치', 'singular', 'plural'],
  'G-033': ['although', 'even though', '양보'],
  'G-034': ['because', 'since', 'as', '원인'],
  'G-035': ['if', 'unless', '조건'],
  'P7-091': ['similar', '유사어', '혼동'],
  'P7-097': ['paraphrase', '바꿔 표현'],
  'LC-033': ['indirect', '우회답변'],
  'M-001': ['time over', '시간 초과'],
}

export function ruleBasedTag(
  questionText: string,
  explanation: string
): string[] {
  const text = (questionText + ' ' + explanation).toLowerCase()
  const matchedCodes: string[] = []

  for (const [tagCode, keywords] of Object.entries(KEYWORD_RULES)) {
    if (keywords.some(kw => text.includes(kw.toLowerCase()))) {
      matchedCodes.push(tagCode)
    }
  }

  return matchedCodes
}
```

export async function autoTagQuestion(questionId: string): Promise<void> {
  // 1. 문제 데이터 조회
  const { data: question } = await supabase
    .from('questions')
    .select('question_text, correct_answer, explanation, part_number')
    .eq('id', questionId)
    .single()

  if (!question) return

  // 2. 이미 태깅 된 경우 스킵
  const { count } = await supabase
    .from('question_tags')
    .select('*', { count: 'exact', head: true })
    .eq('question_id', questionId)

  if ((count ?? 0) > 0) return

  // 3. 1차: 룰 기반
  const ruleCodes = ruleBasedTag(question.question_text, question.explanation ?? '')

  let primaryCode = ruleCodes[0] ?? null
  let secondaryCodes = ruleCodes.slice(1)
  let difficulty = 1

  // 4. 2차: LLM (룰 미탐지 또는 보완)
  if (!primaryCode || secondaryCodes.length === 0) {
    try {
      const { data: tagList } = await supabase
        .from('tag_master')
        .select('tag_code, tag_name')
        .eq('is_active', true)
        .eq('part', `P${question.part_number}`)

      const result = await llmAutoTag(
        question.question_text,
        question.correct_answer,
        question.explanation ?? '',
        question.part_number,
        tagList ?? []
      )

      primaryCode = primaryCode ?? result.primary_tag
      secondaryCodes = [...secondaryCodes, ...result.secondary_tags]
      difficulty = result.difficulty
    } catch (e) {
      console.error('LLM 태깅 실패:', e)
    }
  }

  if (!primaryCode) return

  // 5. tag_code → UUID 변환
  const allCodes = [primaryCode, ...secondaryCodes]
  const { data: tags } = await supabase
    .from('tag_master')
    .select('id, tag_code')
    .in('tag_code', allCodes)

  if (!tags || tags.length === 0) return

  // 6. question_tags 저장
  const inserts = tags.map(tag => ({
    question_id: questionId,
    tag_id: tag.id,
    is_primary: tag.tag_code === primaryCode
  }))

  await supabase.from('question_tags').insert(inserts)

  // 7. 난이도 업데이트 (questions 테이블에 difficulty 컬럼 있을 경우)
  await supabase
    .from('questions')
    .update({ difficulty })
    .eq('id', questionId)
}

interface CoachInput {
  userId: string
  weekNumber: number
  accuracyChange: number       // 전주 대비 정확도 변화율 (%)
  part7TimeChange: number      // 전주 대비 Part7 시간 변화 (초)
  top3WeakTags: WeakTag[]
  predictedScore: number
  predictedScoreChange: number
}

type CoachCase =
  | 'accuracy_down_time_up'    // 정확도 하락 + 시간 증가
  | 'accuracy_up_time_down'    // 정확도 상승 + 시간 단축
  | 'accuracy_up_time_up'      // 정확도 상승 but 시간 증가
  | 'accuracy_down_time_down'  // 정확도 하락 but 시간 단축
  | 'stable'                   // 큰 변화 없음

function detectCoachCase(input: CoachInput): CoachCase {
  const accDown = input.accuracyChange < -2
  const accUp   = input.accuracyChange > 2
  const timeUp  = input.part7TimeChange > 5
  const timeDown = input.part7TimeChange < -5

  if (accDown && timeUp)   return 'accuracy_down_time_up'
  if (accUp && timeDown)   return 'accuracy_up_time_down'
  if (accUp && timeUp)     return 'accuracy_up_time_up'
  if (accDown && timeDown) return 'accuracy_down_time_down'
  return 'stable'
}

export async function buildWeeklyTagPlans(userId: string): Promise<WeekTagPlan[]> {
  // 1. 전체 약점 Top10 조회
  const weakTags = await getTop10WeakTags(userId)

  if (weakTags.length === 0) {
    // 약점 데이터 없을 경우 기본 플랜 반환
    return getDefaultTagPlans()
  }

  // 2. 카테고리별 분류
  const grammarTags = weakTags.filter(t =>
    ['GRAMMAR', 'VOCAB', 'P5', 'P6'].includes(t.category)
  ).slice(0, 3)

  const p7Tags = weakTags.filter(t => t.category === 'P7').slice(0, 3)
  const lcTags  = weakTags.filter(t => t.category === 'LC').slice(0, 3)
  const metaTags = weakTags.filter(t => t.category === 'META').slice(0, 2)

  // P7 trap 태그 보강 (P7-091 ~ P7-097)
  const { data: trapTags } = await supabase
    .from('tag_master')
    .select('id, tag_code')
    .gte('tag_code', 'P7-091')
    .lte('tag_code', 'P7-097')
    .limit(2)

  // 3. Week별 배정
  const plans: WeekTagPlan[] = [
    // Week 1-2: RC 집중
    { week: 1, focusTagIds: grammarTags.map(t => t.tagId),
      focusTagCodes: grammarTags.map(t => t.tagCode),
      dailyTarget: 25, difficultyMix: '2:1:1' },
    { week: 2, focusTagIds: grammarTags.map(t => t.tagId),
      focusTagCodes: grammarTags.map(t => t.tagCode),
      dailyTarget: 25, difficultyMix: '1:2:1' },

    { week: 3, focusTagIds: [...p7Tags.map(t => t.tagId), ...(trapTags?.map(t => t.id) ?? [])],
      focusTagCodes: [...p7Tags.map(t => t.tagCode), ...(trapTags?.map(t => t.tag_code) ?? [])],

    { week: 4, focusTagIds: [...p7Tags.map(t => t.tagId), ...(trapTags?.map(t => t.id) ?? [])],
      focusTagCodes: [...p7Tags.map(t => t.tagCode), ...(trapTags?.map(t => t.tag_code) ?? [])],

    { week: 5, focusTagIds: lcTags.map(t => t.tagId),
      focusTagCodes: lcTags.map(t => t.tagCode),
      dailyTarget: 25, difficultyMix: '2:1:1' },
    { week: 6, focusTagIds: lcTags.map(t => t.tagId),
      focusTagCodes: lcTags.map(t => t.tagCode),
