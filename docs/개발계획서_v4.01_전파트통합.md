```javascript
function calculateTimePenalty(avgTime, benchmarkTime) {
  const timeExcess = (avgTime - benchmarkTime) / benchmarkTime
  
  if (timeExcess <= 0) return 0
  if (timeExcess < 0.05) return -10
  if (timeExcess < 0.10) return -20
  if (timeExcess >= 0.20) return -35
  
  // 선형 보간
  return -10 - (timeExcess * 250)
}
```

### 5. 가중 정확도 계산
```javascript
function calculateWeightedAccuracy(userStats) {
  let totalWeight = 0
  let weightedSum = 0
  
  for (let part = 1; part <= 7; part++) {
    const accuracy = userStats.partAccuracy[part]
    const weight = PART_WEIGHTS[part]
    
    weightedSum += accuracy * weight
    totalWeight += weight
  }
  
  return weightedSum / totalWeight
}
```

```javascript
function calculateWeakTagPenalty(tagStats) {
  let penalty = 0
  
  for (const [tag, wrongRate] of Object.entries(tagStats)) {
    if (wrongRate < 0.4) continue  // 오답률 40% 미만은 무시
    
    const weight = PART5_TAG_WEIGHTS[tag] || PART7_WEIGHTS[tag] || 1.0
    penalty += wrongRate * weight * 5
  }
  
  return Math.round(penalty)
}
```

```javascript
  // 2. 가중 정확도 계산
  const weightedAccuracy = calculateWeightedAccuracy(stats)
  
  // 3. 기본 점수
  const baseScore = weightedAccuracy * 990
  
  // 4. 약점 태그 패널티
  const weakTagPenalty = calculateWeakTagPenalty(stats.tagWrongRates)
  
  // 5. Part 7 시간 패널티
  const timePenalty = calculateTimePenalty(
    stats.avgPart7Time,
    90  // 벤치마크: 90초 (1분 30초)
  )
  
  // 6. 최종 점수
  const estimatedScore = Math.round(
    baseScore - weakTagPenalty + timePenalty
  )
  
  // 7. 상승률 계산 (최근 2주)
  const improvementRate = await calculateImprovementRate(userId, 14)
  
  // 8. 2주 후 예상 점수
  const predictedNextScore = Math.round(
    estimatedScore + (improvementRate * 10)
  )
  
  // 9. 목표까지 남은 점수
  const targetGap = user.targetScore - estimatedScore
  
  // 10. score_analytics에 저장
  await supabase
    .from('score_analytics')
    .upsert({
      user_id: userId,
      estimated_score: estimatedScore,
      weighted_accuracy: weightedAccuracy,
      weak_tag_penalty: weakTagPenalty,
      time_penalty: timePenalty,
      improvement_rate: improvementRate,
      predicted_next_score: predictedNextScore,
      target_gap: targetGap,
      last_analyzed_at: new Date()
    })
  
  return {
    estimatedScore,
    predictedNextScore,
    targetGap,
    improvementRate
  }
}
```

### 8. 상승률 계산 함수
```javascript
async function calculateImprovementRate(userId, days) {
  // 최근 N일간의 정확도 데이터 조회
  const { data } = await supabase
    .from('wrong_notes')
    .select('attempt_date, part')
    .eq('user_id', userId)
    .gte('attempt_date', new Date(Date.now() - days * 24 * 60 * 60 * 1000))
    .order('attempt_date')
  
  if (!data || data.length < 10) return 0
  
  // 일별 정확도 계산
  const dailyAccuracy = {}
  
  data.forEach(note => {
    const date = note.attempt_date.split('T')[0]
    if (!dailyAccuracy[date]) {
      dailyAccuracy[date] = { total: 0, wrong: 0 }
    }
    dailyAccuracy[date].total++
    dailyAccuracy[date].wrong++
  })
  
  // 선형 회귀로 추세 계산
  const dates = Object.keys(dailyAccuracy).sort()
  const accuracies = dates.map(date => {
    const day = dailyAccuracy[date]
    return 1 - (day.wrong / day.total)
  })
  
  const slope = linearRegression(accuracies)
  
  // 일일 상승률 → 2주 상승률로 변환
  return slope * 14
}
```
