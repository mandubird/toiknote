```sql
CREATE OR REPLACE FUNCTION generate_referral_code()
RETURNS TEXT AS $$
DECLARE
  code TEXT;
  exists BOOLEAN;
BEGIN
  LOOP
    -- TOKAI- + 4자리 랜덤
    code := 'TOKAI-' || upper(substring(md5(random()::text) from 1 for 4));
    
    SELECT EXISTS(SELECT 1 FROM users WHERE referral_code = code) INTO exists;
    
    EXIT WHEN NOT exists;
  END LOOP;
  
  RETURN code;
END;
$$ LANGUAGE plpgsql;

-- 신규 사용자 생성 시 자동 코드 발급
CREATE OR REPLACE FUNCTION auto_generate_referral_code()
RETURNS TRIGGER AS $$
BEGIN
  IF NEW.referral_code IS NULL THEN
    NEW.referral_code := generate_referral_code();
  END IF;
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER before_user_insert
BEFORE INSERT ON users
FOR EACH ROW
EXECUTE FUNCTION auto_generate_referral_code();
```

```sql
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  
  -- 추천인/피추천인

  referred_user_id UUID REFERENCES users(id) ON DELETE CASCADE,
  
  -- 상태

    CHECK (status IN ('pending', 'completed', 'failed')),
  
  -- 보상 정보

  rewarded_at TIMESTAMPTZ,
  
  -- 메타데이터
  created_at TIMESTAMPTZ DEFAULT NOW(),
  completed_at TIMESTAMPTZ
);

CREATE INDEX idx_referral_logs_referrer ON referral_logs(referrer_user_id);
CREATE INDEX idx_referral_logs_referred ON referral_logs(referred_user_id);
CREATE INDEX idx_referral_logs_status ON referral_logs(status);

-- 중복 방지
CREATE UNIQUE INDEX idx_referral_unique 
ON referral_logs(referrer_user_id, referred_user_id);
```

**상태 전환:**
```
pending:    피추천인 가입 (결제 전)
completed:  피추천인 결제 완료 → 추천인 보상
```

### 3-1. 공유 보상 함수

```javascript
async function rewardShareBonus(userId, platform) {
  // 1. 사용자 조회
  const { data: user } = await supabase
    .from('users')
    .select('*')
    .eq('id', userId)
    .single()
  
  // 2. 보너스 한도 체크
  if (user.bonus_days >= BONUS_RULES.maxBonusDays) {
    throw new Error('최대 보너스 한도에 도달했습니다')
  }
  
  // 3. 공유 횟수 체크
  if (user.share_reward_count >= BONUS_RULES.shareMaxCount) {
    throw new Error('공유 보상은 최대 3회까지 받을 수 있습니다')
  }
  
  // 4. 하루 1회 제한 체크
  const today = new Date().toISOString().split('T')[0]
  const { data: todayShare } = await supabase
    .from('share_logs')
    .select('id')
    .eq('user_id', userId)
    .gte('created_at', today)
    .single()
  
  if (todayShare) {
    throw new Error('하루 1회만 공유 보상을 받을 수 있습니다')
  }
  
  // 5. share_logs 기록
  await supabase
    .from('share_logs')
    .insert({
      user_id: userId,
      share_type: 'diagnosis_share',
      platform: platform,
      reward_days: BONUS_RULES.shareReward,
      rewarded: true
    })
  
  // 6. 보너스 적용
  const newEndDate = new Date(user.program_end_date)
  newEndDate.setDate(newEndDate.getDate() + BONUS_RULES.shareReward)
  
  await supabase
    .from('users')
    .update({
      program_end_date: newEndDate,
      bonus_days: user.bonus_days + BONUS_RULES.shareReward,
      share_reward_count: user.share_reward_count + 1,
      last_shared_at: new Date()
    })
    .eq('id', userId)
  
  return {
    success: true,
    rewardDays: BONUS_RULES.shareReward,
    newEndDate: newEndDate,
    remainingShares: BONUS_RULES.shareMaxCount - (user.share_reward_count + 1)
  }
}
```

### 3-2. 추천 가입 기록 함수

```javascript
async function recordReferral(referralCode, newUserId) {
  // 1. 추천인 조회
  const { data: referrer } = await supabase
    .from('users')
    .select('id')
    .eq('referral_code', referralCode)
    .single()
  
  if (!referrer) {
    throw new Error('유효하지 않은 추천 코드입니다')
  }
  
  // 2. 자가 추천 방지
  if (referrer.id === newUserId) {
    throw new Error('자신을 추천할 수 없습니다')
  }
  
  // 3. 피추천인 users 업데이트
  await supabase
    .from('users')
    .update({ referred_by: referralCode })
    .eq('id', newUserId)
  
  // 4. referral_logs에 pending 상태로 기록
  await supabase
    .from('referral_logs')
    .insert({
      referrer_user_id: referrer.id,
      referred_user_id: newUserId,
      status: 'pending'
    })
  
  console.log('추천 관계 기록 완료 (결제 대기)')
}
```

### 3-3. 결제 완료 시 추천 보상 지급

```javascript
async function completeReferralReward(newUserId) {
  // 1. referral_logs 조회
  const { data: log } = await supabase
    .from('referral_logs')
    .select('*, referrer:referrer_user_id(*)')
    .eq('referred_user_id', newUserId)
    .eq('status', 'pending')
    .single()
  
  if (!log) {
    console.log('추천 관계 없음')
    return
  }
  
  const referrer = log.referrer
  
  // 2. 보너스 한도 체크
  if (referrer.bonus_days >= BONUS_RULES.maxBonusDays) {
    console.log('추천인이 보너스 한도 도달')
    return
  }
  
  // 3. 기본 보상
  let rewardDays = BONUS_RULES.referralReward  // 7일
  
  // 4. 누적 보너스 계산
  const newReferralCount = referrer.referral_count + 1
  
  if (newReferralCount === 3) {
    rewardDays += BONUS_RULES.referralBonusAt3  // +21일
  } else if (newReferralCount === 5) {
    rewardDays += BONUS_RULES.referralBonusAt5  // +35일
  }
  
  // 5. 한도 초과 방지
  const maxAllowed = BONUS_RULES.maxBonusDays - referrer.bonus_days
  rewardDays = Math.min(rewardDays, maxAllowed)
  
  // 6. 보너스 적용
  const newEndDate = new Date(referrer.program_end_date)
  newEndDate.setDate(newEndDate.getDate() + rewardDays)
  
  await supabase
    .from('users')
    .update({
      program_end_date: newEndDate,
      bonus_days: referrer.bonus_days + rewardDays,
      referral_count: newReferralCount
    })
    .eq('id', referrer.id)
  
  // 7. referral_logs 완료 처리
  await supabase
    .from('referral_logs')
    .update({
      status: 'completed',
      reward_days: rewardDays,
      rewarded_at: new Date(),
      completed_at: new Date()
    })
    .eq('id', log.id)
  
  console.log(`추천 보상: ${rewardDays}일 지급 완료`)
}
```

### 3-4. 보너스 계산식 (요약)

```javascript
function calculateBonus(user, rewardDays) {
  if (user.bonus_days + rewardDays > user.max_bonus_days) {
    rewardDays = user.max_bonus_days - user.bonus_days
  }
  
  if (rewardDays <= 0) {
    return { canReward: false }
  }
  
  const newEndDate = new Date(user.program_end_date)
  newEndDate.setDate(newEndDate.getDate() + rewardDays)
  
  return {
    canReward: true,
    rewardDays: rewardDays,
    newEndDate: newEndDate,
    newBonusDays: user.bonus_days + rewardDays
  }
}
```

### 5-2. 자동 뱃지 부여

```javascript
async function autoAwardBadge(userId) {
  // 1. 점수 조회
  const { data: analytics } = await supabase
    .from('score_analytics')
    .select('estimated_score')
    .eq('user_id', userId)
    .single()
  
  const score = analytics.estimated_score
  
  // 2. 뱃지 등급 결정
  let newBadgeLevel = 'none'
  
  if (score >= 880) {
    newBadgeLevel = 'elite'
  } else if (score >= 850) {
    newBadgeLevel = 'challenger'
  }
  
  if (newBadgeLevel === 'none') return
  
  // 3. 현재 뱃지 조회
  const { data: user } = await supabase
    .from('users')
    .select('badge_level')
    .eq('id', userId)
    .single()
  
  // 4. 업그레이드만 허용
  const levels = ['none', 'challenger', 'elite', '900']
  const currentLevel = levels.indexOf(user.badge_level)
  const targetLevel = levels.indexOf(newBadgeLevel)
  
  if (targetLevel <= currentLevel) return
  
  // 5. 뱃지 업데이트
  await supabase
    .from('users')
    .update({ badge_level: newBadgeLevel })
    .eq('id', userId)
  
  console.log(`뱃지 부여: ${newBadgeLevel}`)
}
```

```javascript
async function checkCardAbuse(cardNumber, userId) {
  const cardHash = crypto
    .createHash('sha256')
    .update(cardNumber)
    .digest('hex')
  
  const { count } = await supabase
    .from('payment_fingerprints')
    .select('*', { count: 'exact' })
    .eq('card_hash', cardHash)
  
  if (count >= 5) {
    throw new Error('동일한 카드로는 최대 5개 계정까지만 결제 가능합니다')
  }
  
  await supabase
    .from('payment_fingerprints')
    .insert({ user_id: userId, card_hash: cardHash })
}
```

## 📊 성장 시뮬레이션

### 바이럴 계수 계산

```javascript
const VIRAL_ASSUMPTIONS = {
  shareRate: 0.40,           // 40% 사용자가 공유
  referralRate: 0.30,        // 30% 사용자가 추천
  referralSignupRate: 0.20,  // 추천 링크 20% 가입
  referralPaymentRate: 0.40  // 가입자 40% 결제
}

// 바이럴 계수 k
const k = 
  (VIRAL_ASSUMPTIONS.shareRate * 0.05) +  // 공유 전환
  (VIRAL_ASSUMPTIONS.referralRate * 
   VIRAL_ASSUMPTIONS.referralSignupRate * 
   VIRAL_ASSUMPTIONS.referralPaymentRate)  // 추천 전환

// k ≈ 0.02 + 0.024 = 0.044
```

### 성장 예측
