### 1-4. 시간 가중치 (Part 7 전용)

```javascript
function calculateTimePenalty(avgTime, benchmarkTime = 50) {
  const timeExcess = (avgTime - benchmarkTime) / benchmarkTime
  
  if (timeExcess <= 0) {
    return 0  // 시간 내 완료
  }
  
  if (timeExcess < 0.05) {
    return -10  // 5% 초과
  }
  
  if (timeExcess < 0.10) {
    return -20  // 10% 초과
  }
  
  if (timeExcess >= 0.20) {
    return -35  // 20% 이상 초과
  }
  
  // 선형 보간
  return Math.round(-10 - (timeExcess * 250))
}
```

**예시:**
```
기준 시간: 50분

실제 시간 52분 (5% 초과)   → -10점
실제 시간 55분 (10% 초과)  → -20점
실제 시간 60분 (20% 초과)  → -35점
```

### 2-1. 가중 정확도 계산

```javascript
function calculateWeightedAccuracy(partAccuracy) {
  let totalWeight = 0
  let weightedSum = 0
  
  for (let part = 1; part <= 7; part++) {
    const accuracy = partAccuracy[part]  // 0.0 ~ 1.0
    const weight = PART_WEIGHTS[part]
    
    weightedSum += accuracy * weight
    totalWeight += weight
  }
  
  return weightedSum / totalWeight
}
```

**예시:**
```
입력:
{
  1: 0.85,  // Part 1: 85%
  2: 0.78,
  3: 0.82,
  4: 0.75,
  5: 0.88,
  6: 0.83,
  7: 0.72   // Part 7: 72%
}

계산:
(0.85×1.0 + 0.78×1.0 + 0.82×1.1 + 0.75×1.1 + 
 0.88×1.2 + 0.83×1.1 + 0.72×1.4) / 7.8

= 0.806 (80.6%)
```

### 2-2. 태그 보정 점수 (약점 패널티)

```javascript
function calculateWeakTagPenalty(tagStats, tagWeights) {
  let penalty = 0
  
  Object.entries(tagStats).forEach(([tag, wrongRate]) => {
    // 오답률 40% 이상만 패널티 적용
    if (wrongRate < 0.4) return
    
    const weight = tagWeights[tag] || 1.0
    penalty += wrongRate * weight * 5
  })
  
  return Math.round(penalty)
}
```

**예시:**
```
태그별 오답률:
{
  '시제': 0.48,         // 48% (40% 이상)
  '관계대명사': 0.42,   // 42% (40% 이상)
  '접속사': 0.35,       // 35% (40% 미만 → 무시)
  '어휘': 0.50          // 50% (40% 이상)
}

계산:
시제:         0.48 × 1.3 × 5 = 3.12
관계대명사:   0.42 × 1.4 × 5 = 2.94
어휘:         0.50 × 1.1 × 5 = 2.75

총 패널티: 8.81 ≈ 9점
```

```javascript
async function calculateEstimatedScore(userId) {
  // 1. 사용자 데이터 조회
  const partAccuracy = await getPartAccuracy(userId)
  const tagStats = await getTagStats(userId)
  const part7AvgTime = await getPart7AvgTime(userId)
  
  // 2. 가중 정확도 계산
  const weightedAccuracy = calculateWeightedAccuracy(partAccuracy)
  
  // 3. 기본 점수
  const baseScore = weightedAccuracy * 990  // 990 만점
  
  // 4. 태그 가중치 조회
  const { data: weights } = await supabase
    .from('tag_weights')
    .select('*')
  
  const tagWeightMap = {}
  weights.forEach(w => {
    tagWeightMap[w.tag_name] = w.weight
  })
  
  // 5. 약점 패널티
  const weakTagPenalty = calculateWeakTagPenalty(tagStats, tagWeightMap)
  
  // 6. 시간 패널티
  const timePenalty = getTimePenalty(part7AvgTime)
  
  // 7. 최종 점수
  const estimatedScore = Math.round(
    baseScore - weakTagPenalty + timePenalty  // timePenalty는 음수
  )
  
  // 8. score_analytics에 저장
  await supabase
    .from('score_analytics')
    .upsert({
      user_id: userId,
      estimated_score: estimatedScore,
      weighted_accuracy: weightedAccuracy,
      weak_tag_penalty: weakTagPenalty,
      time_penalty: timePenalty
    })
  
  return {
    estimatedScore,
    baseScore: Math.round(baseScore),
    weakTagPenalty,
    timePenalty,
    weightedAccuracy
  }
}
```

### 2-5. 상승 예측 모델

```javascript
async function predictScoreImprovement(userId) {
  // 1. 최근 2주 데이터 조회
  const twoWeeksAgo = new Date()
  twoWeeksAgo.setDate(twoWeeksAgo.getDate() - 14)
  
  const { data: recentAccuracy } = await supabase
    .from('solved_questions')
    .select('attempt_date, is_correct, part')
    .eq('user_id', userId)
    .gte('attempt_date', twoWeeksAgo.toISOString())
    .order('attempt_date')
  
  // 2. 일별 정확도 계산
  const dailyAccuracy = {}
  recentAccuracy.forEach(q => {
    const date = q.attempt_date.split('T')[0]
    if (!dailyAccuracy[date]) {
      dailyAccuracy[date] = { total: 0, correct: 0 }
    }
    dailyAccuracy[date].total++
    if (q.is_correct) dailyAccuracy[date].correct++
  })
  
  // 3. 선형 회귀로 상승률 계산
  const dates = Object.keys(dailyAccuracy).sort()
  const accuracies = dates.map(date => {
    const day = dailyAccuracy[date]
    return day.correct / day.total
  })
  
  const slope = calculateLinearRegression(accuracies)
  
  // 4. 2주 후 예상 점수
  const currentScore = await calculateEstimatedScore(userId)
  const improvementRate = slope * 14  // 14일 상승률
  
  const predictedNextScore = Math.round(
    currentScore.estimatedScore + (improvementRate * 100)
  )
  
  return {
    currentScore: currentScore.estimatedScore,
    predictedNextScore: predictedNextScore,
    improvementRate: (improvementRate * 100).toFixed(1),
    targetGap: 900 - currentScore.estimatedScore
  }
}

// 선형 회귀 함수
function calculateLinearRegression(values) {
  const n = values.length
  const sumX = (n * (n - 1)) / 2
  const sumY = values.reduce((a, b) => a + b, 0)
  const sumXY = values.reduce((sum, y, x) => sum + x * y, 0)
  const sumX2 = (n * (n - 1) * (2 * n - 1)) / 6
  
  return (n * sumXY - sumX * sumY) / (n * sumX2 - sumX * sumX)
}
```

```javascript
  targetGap: 88
```

### KPI 추적

```sql
CREATE TABLE conversion_events (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  event_type TEXT,  -- 'page_view', 'diagnosis_start', 'diagnosis_complete', 'purchase_click', 'purchase_complete'
  user_id UUID,
  session_id TEXT,
  
  -- 이벤트 데이터
  event_data JSONB,
  
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 일별 전환율 조회
SELECT 
  DATE(created_at) as date,
  COUNT(*) FILTER (WHERE event_type = 'page_view') as page_views,
  COUNT(*) FILTER (WHERE event_type = 'diagnosis_start') as diagnosis_starts,
  COUNT(*) FILTER (WHERE event_type = 'diagnosis_complete') as diagnosis_completes,
  COUNT(*) FILTER (WHERE event_type = 'purchase_complete') as purchases,
  
  ROUND(
    COUNT(*) FILTER (WHERE event_type = 'diagnosis_start')::numeric / 
    NULLIF(COUNT(*) FILTER (WHERE event_type = 'page_view'), 0) * 100,
    2
  ) as diagnosis_rate,
  
  ROUND(
    COUNT(*) FILTER (WHERE event_type = 'purchase_complete')::numeric / 
    NULLIF(COUNT(*) FILTER (WHERE event_type = 'diagnosis_complete'), 0) * 100,
    2
  ) as purchase_rate

FROM conversion_events
WHERE created_at >= NOW() - INTERVAL '30 days'
GROUP BY DATE(created_at)
ORDER BY date DESC;
```
