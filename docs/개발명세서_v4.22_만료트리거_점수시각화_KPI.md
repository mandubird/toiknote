```typescript
// ⚠️ 기존 Cron 4개 아래에 추가만 할 것. 기존 코드 수정 금지.

// 5. 매일 오전 9시 — 만료 임박 경고 플래그 설정
schedule.scheduleJob('0 9 * * *', async () => {
  await checkExpiryWarning()
})

async function checkExpiryWarning(): Promise<void> {
  const now = new Date()

  // D-7, D-3, D-1 대상자 조회
  const targets = [7, 3, 1]

  for (const daysLeft of targets) {
    const targetDate = new Date(now)
    targetDate.setDate(targetDate.getDate() + daysLeft)
    const dateStr = targetDate.toISOString().split('T')[0]

    const { data: users } = await supabase
      .from('users')
      .select('id, program_end_date, expiry_warning_sent_at')
      .eq('program_status', 'active')
      .gte('program_end_date', dateStr + 'T00:00:00Z')
      .lt('program_end_date', dateStr + 'T23:59:59Z')

    if (!users || users.length === 0) continue

    for (const user of users) {
      // 이미 오늘 발송한 경우 스킵
      if (user.expiry_warning_sent_at) {
        const lastSent = new Date(user.expiry_warning_sent_at)
        const isSameDay = lastSent.toISOString().split('T')[0] === now.toISOString().split('T')[0]
        if (isSameDay) continue
      }

      // 플래그 업데이트 (인앱 배너가 이 값을 읽어서 표시)
      await supabase
        .from('users')
        .update({ expiry_warning_sent_at: now })
        .eq('id', user.id)
    }
  }

  console.log('[Cron] checkExpiryWarning 완료')
}
```

### 2-3. 만료 D-Day 계산 유틸 함수

```typescript
// src/utils/expiryUtils.ts (신규 파일)

export function getDaysUntilExpiry(programEndDate: string): number {
  const end = new Date(programEndDate)
  const now = new Date()
  const diff = end.getTime() - now.getTime()
  return Math.ceil(diff / (1000 * 60 * 60 * 24))
}

export type ExpiryWarningLevel = 'none' | 'warning' | 'urgent' | 'critical'

export function getExpiryWarningLevel(daysLeft: number): ExpiryWarningLevel {
  if (daysLeft > 7)  return 'none'
  if (daysLeft >= 4) return 'warning'   // D-7 ~ D-4
  if (daysLeft >= 2) return 'urgent'    // D-3 ~ D-2
  if (daysLeft >= 0) return 'critical'  // D-1 ~ D-0
  return 'none'
}
```

```typescript
export async function extendProgram(userId: string, days: number = 30): Promise<{
  success: boolean
  newEndDate: Date
}> {
  const { data: user } = await supabase
    .from('users')
    .select('program_end_date, program_status')
    .eq('id', userId)
    .single()

  if (!user) throw new Error('사용자를 찾을 수 없습니다.')

  const baseDate = user.program_status === 'expired'
    ? new Date()
    : new Date(user.program_end_date)

  newEndDate.setDate(newEndDate.getDate() + days)

    .from('users')

      program_status: 'active',
      expiry_warning_sent_at: null  // 연장 시 경고 플래그 초기화

    .eq('id', userId)

  return { success: true, newEndDate }
```

## 📊 3. 900까지 남은 점수 시각화 (ScoreGaugeCard)

### 3-1. 컴포넌트 설계 원칙

```
목표: 사용자가 "얼마나 더 올려야 하는지" 직관적으로 인식
방식: 가로 프로그레스 바 + 숫자 강조
색상 구간:
  - 50점 이내  → 초록 (목표 코앞)
  - 100점 이내 → 노랑 (중간)
  - 100점 초과 → 빨강 (갈 길 멀음)
```

### 3-2. ScoreGaugeCard 컴포넌트

```tsx
// src/components/ScoreGaugeCard.tsx (신규 파일)

interface ScoreGaugeCardProps {
  currentScore: number       // 현재 예상 점수
  targetScore?: number       // 목표 점수 (기본값 900)
  maxScore?: number          // 최대 점수 (기본값 990)
}

export function ScoreGaugeCard({
  currentScore,
  targetScore = 900,
  maxScore = 990
}: ScoreGaugeCardProps) {
  const gap = Math.max(0, targetScore - currentScore)
  const progress = Math.min(100, (currentScore / targetScore) * 100)

  // 색상 구간
  const getGaugeColor = () => {
    if (gap <= 50)  return { bar: '#22c55e', text: '#16a34a', label: '거의 다 왔어요!' }
    if (gap <= 100) return { bar: '#f59e0b', text: '#d97706', label: '조금만 더!' }
    return            { bar: '#ef4444', text: '#dc2626', label: '집중 훈련이 필요해요' }
  }

  const color = getGaugeColor()

  return (
    <div className="score-gauge-card">
      {/* 헤더 */}
      <div className="gauge-header">
        <span className="gauge-title">900까지 남은 점수</span>
        <span className="gauge-label" style={{ color: color.text }}>
          {color.label}
        </span>
      </div>

      {/* 점수 표시 */}
      <div className="score-display">
        <div className="score-item">
          <span className="score-label">현재</span>
          <strong className="score-value">{currentScore}점</strong>
        </div>
        <div className="score-arrow">→</div>
        <div className="score-item target">
          <span className="score-label">목표</span>
          <strong className="score-value target">{targetScore}점</strong>
        </div>
        <div className="score-gap-box">
          <span className="gap-label">남은 점수</span>
          <strong className="gap-value" style={{ color: color.text }}>
            -{gap}점
          </strong>
        </div>
      </div>

      {/* 프로그레스 바 */}
      <div className="gauge-bar-container">
        <div
          className="gauge-bar-fill"
          style={{
            width: `${progress}%`,
            backgroundColor: color.bar,
            transition: 'width 0.8s ease-in-out'
          }}
        />
        {/* 목표 마커 */}
        <div
          className="gauge-target-marker"
          style={{ left: `${(targetScore / maxScore) * 100}%` }}
        >
          <span>900</span>
        </div>
      </div>

      {/* 하단 범례 */}
      <div className="gauge-footer">
        <span>200</span>
        <span>990</span>
      </div>
    </div>
  )
}
```

```typescript
    referralsResult,
    reviewsResult,
    badgesResult
  ] = await Promise.all([
    // 사용자 상태별 수
    supabase
      .from('users')
      .select('program_status, badge_level'),

    // 오늘 결제 (subscriptions 또는 payments 테이블 참조)
    supabase
      .from('subscriptions')
      .select('amount, created_at')
      .gte('created_at', today + 'T00:00:00Z'),

    // 이번 달 결제
    supabase
      .from('subscriptions')
      .select('amount, created_at')
      .gte('created_at', monthStart),

    // 전체 진단 수
    supabase
      .from('diagnostic_results')
      .select('id', { count: 'exact', head: true }),

    // 공유 로그
    supabase
      .from('share_logs')
      .select('id', { count: 'exact', head: true }),

    // 추천 로그
    supabase
      .from('referral_logs')
      .select('status'),

    // 후기
    supabase
      .from('reviews')
      .select('approved, rating'),

    // 배지 (users에서 집계)
    supabase
      .from('users')
      .select('badge_level')
      .neq('badge_level', 'none')
  ])

  const users = usersResult.data ?? []
  const payments = paymentsResult.data ?? []
  const paymentsMonth = paymentsMonthResult.data ?? []
  const referrals = referralsResult.data ?? []
  const reviews = reviewsResult.data ?? []
  const badges = badgesResult.data ?? []

  // 매출 계산 (pricing_plans 참조 대신 subscriptions.amount 합산)
  const todayRevenue = payments.reduce((sum, p) => sum + (p.amount ?? 0), 0)
  const monthRevenue = paymentsMonth.reduce((sum, p) => sum + (p.amount ?? 0), 0)

  // 사용자 상태 집계
  const activeUsers = users.filter(u => u.program_status === 'active').length
  const expiredUsers = users.filter(u => u.program_status === 'expired').length
  const completedUsers = users.filter(u => u.program_status === 'completed').length

  // 퍼널 전환율
  const totalDiagnoses = diagnosticsResult.count ?? 0
  const totalPayments = paymentsMonth.length
  const diagnosisToPaymentRate = totalDiagnoses > 0
    ? Math.round((totalPayments / totalDiagnoses) * 100 * 10) / 10
    : 0

  // 바이럴
  const completedReferrals = referrals.filter(r => r.status === 'completed').length
  const referralConversionRate = referrals.length > 0
    ? Math.round((completedReferrals / referrals.length) * 100 * 10) / 10
    : 0

  // 후기
  const approvedReviews = reviews.filter(r => r.approved).length
  const pendingReviews = reviews.filter(r => !r.approved).length
  const avgRating = approvedReviews > 0
    ? Math.round(
        reviews.filter(r => r.approved).reduce((sum, r) => sum + r.rating, 0) / approvedReviews * 10
      ) / 10
    : 0

  // 배지
  const challengerCount = badges.filter(u => u.badge_level === 'challenger').length
  const eliteCount = badges.filter(u => u.badge_level === 'elite').length
  const badge900Count = badges.filter(u => u.badge_level === '900').length

  return {
    todayRevenue,
    monthRevenue,
    todayPayments: payments.length,
    monthPayments: totalPayments,
    activeUsers,
    expiredUsers,
    completedUsers,
    totalUsers: users.length,
    totalDiagnoses,
    totalPayments,
    diagnosisToPaymentRate,
    totalShares: sharesResult.count ?? 0,
    totalReferrals: referrals.length,
    completedReferrals,
    referralConversionRate,
    pendingReviews,
    approvedReviews,
    avgRating,
    challengerCount,
    eliteCount,
    badge900Count
  }
}
```
