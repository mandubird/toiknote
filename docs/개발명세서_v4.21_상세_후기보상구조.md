## 🗄️ 1. DB 설계

### 1-1. reviews 테이블 (신규)

```sql
CREATE TABLE reviews (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),

  -- 작성자
  user_id UUID REFERENCES users(id) ON DELETE CASCADE NOT NULL,
  nickname TEXT NOT NULL,                   -- 실명 아닌 닉네임 표시용

  -- 점수 정보
  score_before INTEGER CHECK (score_before BETWEEN 200 AND 990),
  score_after  INTEGER CHECK (score_after  BETWEEN 200 AND 990),

  -- 후기 내용
  helpful_feature TEXT,                     -- "가장 도움된 기능" 선택형
  content TEXT NOT NULL CHECK (char_length(content) >= 200),
  rating INTEGER NOT NULL CHECK (rating BETWEEN 1 AND 5),

  -- 승인 / 보상
  approved BOOLEAN DEFAULT FALSE,
  rewarded BOOLEAN DEFAULT FALSE,           -- +5일 지급 완료 여부

  -- 관리자
  reviewed_by UUID REFERENCES users(id),
  reviewed_at TIMESTAMPTZ,
  reject_reason TEXT,

  -- 메타데이터
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE INDEX idx_reviews_user_id   ON reviews(user_id);
CREATE INDEX idx_reviews_approved  ON reviews(approved);
CREATE INDEX idx_reviews_rating    ON reviews(rating);
CREATE INDEX idx_reviews_created   ON reviews(created_at DESC);

-- 사용자당 1개 후기만 허용
CREATE UNIQUE INDEX idx_reviews_user_unique ON reviews(user_id);
```

**RLS 정책:**
```sql
-- 승인된 후기만 누구나 조회 가능
ALTER TABLE reviews ENABLE ROW LEVEL SECURITY;

CREATE POLICY "approved reviews visible to all"
  ON reviews FOR SELECT
  USING (approved = TRUE);

-- 본인 후기는 본인만 조회/삽입 가능
CREATE POLICY "own review insert"
  ON reviews FOR INSERT
  WITH CHECK (auth.uid() = user_id);

CREATE POLICY "own review select"
  ON reviews FOR SELECT
  USING (auth.uid() = user_id);

-- 관리자는 전체 조회/수정 가능 (is_admin 컬럼 또는 별도 관리자 판별)
CREATE POLICY "admin full access"
  ON reviews FOR ALL
  USING (
    EXISTS (
      SELECT 1 FROM users
      WHERE id = auth.uid() AND is_admin = TRUE
    )
  );
```

## 💡 2. 후기 작성 기능

### 2-1. 작성 가능 조건 검증

```typescript
// src/services/reviewService.ts

import { supabase } from '@/lib/supabase'

export async function canWriteReview(userId: string): Promise<{
  canWrite: boolean
  reason?: string
}> {
  // 1. 이미 후기 있는지 확인
  const { data: existing } = await supabase
    .from('reviews')
    .select('id')
    .eq('user_id', userId)
    .single()

  if (existing) {
    return { canWrite: false, reason: '이미 후기를 작성하셨습니다.' }
  }

  // 2. 사용자 프로그램 상태 확인
  const { data: user } = await supabase
    .from('users')
    .select('program_start_date, program_status, current_week')
    .eq('id', userId)
    .single()

  if (!user || user.program_status === 'none') {
    return { canWrite: false, reason: '프로그램을 시작한 후 작성 가능합니다.' }
  }

  // 3. 14일 이상 사용 여부
  const startDate = new Date(user.program_start_date)
  const today = new Date()
  const daysUsed = Math.floor((today.getTime() - startDate.getTime()) / (1000 * 60 * 60 * 24))

  if (daysUsed < 14) {
    return {
      canWrite: false,
      reason: `프로그램 시작 후 14일 이상 사용 시 작성 가능합니다. (현재 ${daysUsed}일째)`
    }
  }

  // 4. 100문제 이상 풀이 여부
  const { count } = await supabase
    .from('solved_questions')
    .select('*', { count: 'exact', head: true })
    .eq('user_id', userId)

  if ((count ?? 0) < 100) {
    return {
      canWrite: false,
      reason: `최소 100문제 이상 풀이 후 작성 가능합니다. (현재 ${count}문제)`
    }
  }

  return { canWrite: true }
}
```

### 2-2. 후기 저장 함수

```typescript
export interface ReviewInput {
  userId: string
  nickname: string
  scoreBefore: number
  scoreAfter: number
  helpfulFeature: string
  content: string
  rating: number
}

export async function submitReview(input: ReviewInput): Promise<{ success: boolean; message: string }> {
  // 1. 작성 가능 조건 재확인
  const { canWrite, reason } = await canWriteReview(input.userId)
  if (!canWrite) throw new Error(reason)

  // 2. 후기 저장 (approved = false, rewarded = false 기본값)
  const { error } = await supabase
    .from('reviews')
    .insert({
      user_id: input.userId,
      nickname: input.nickname,
      score_before: input.scoreBefore,
      score_after: input.scoreAfter,
      helpful_feature: input.helpfulFeature,
      content: input.content,
      rating: input.rating
    })

  if (error) throw new Error('후기 저장에 실패했습니다: ' + error.message)

  return {
    success: true,
    message: '후기가 제출되었습니다. 관리자 승인 후 게시됩니다. 승인 시 +5일이 지급됩니다.'
  }
}
```

### 3-2. 보상 지급 함수

```typescript
// src/services/reviewService.ts (추가)

export async function rewardReviewBonus(reviewId: string, adminId: string): Promise<void> {
  const REVIEW_REWARD_DAYS = 5

  // 1. 후기 조회
  const { data: review } = await supabase
    .from('reviews')
    .select('*, user:user_id(*)')
    .eq('id', reviewId)
    .single()

  if (!review) throw new Error('후기를 찾을 수 없습니다.')
  if (review.rewarded) throw new Error('이미 보상이 지급된 후기입니다.')
  if (!review.approved) throw new Error('승인되지 않은 후기입니다.')

  const user = review.user

  // 2. bonusDays 한도 체크 (v4.20 규칙 재사용)
  if (user.bonus_days >= user.max_bonus_days) {
    // 보상 한도 초과 시 rewarded만 true로 변경 (지급 불가 처리)
    await supabase
      .from('reviews')
      .update({ rewarded: true })
      .eq('id', reviewId)
    return
  }

  const actualReward = Math.min(
    REVIEW_REWARD_DAYS,
    user.max_bonus_days - user.bonus_days
  )

  // 3. programEndDate 연장
  const newEndDate = new Date(user.program_end_date)
  newEndDate.setDate(newEndDate.getDate() + actualReward)

  // 4. users 업데이트
  await supabase
    .from('users')
    .update({
      program_end_date: newEndDate,
      bonus_days: user.bonus_days + actualReward,
      review_reward_given: true
    })
    .eq('id', user.id)

  // 5. reviews.rewarded = true
  await supabase
    .from('reviews')
    .update({
      rewarded: true,
      reviewed_by: adminId,
      reviewed_at: new Date()
    })
    .eq('id', reviewId)
}
```

```typescript
export async function approveReview(reviewId: string, adminId: string): Promise<void> {
  // 1. approved = true 업데이트

    .from('reviews')
    .update({
      approved: true,

      reviewed_at: new Date(),

    })
    .eq('id', reviewId)
```

```typescript
// src/services/reviewService.ts (추가)

export type ReviewSortType = 'latest' | 'score_gain' | 'rating'

export async function getApprovedReviews(
  sort: ReviewSortType = 'latest',
  page: number = 1,
  limit: number = 12
) {
  let query = supabase
    .from('reviews')
    .select(`
      id,
      nickname,
      score_before,
      score_after,
      helpful_feature,
      content,
      rating,
      created_at,
      user_id,
      users!inner(badge_level)
    `)
    .eq('approved', true)

  // 정렬 적용
  if (sort === 'latest') {
    query = query.order('created_at', { ascending: false })
  } else if (sort === 'score_gain') {
    // score_after - score_before 내림차순 (Supabase에서 computed sort는 직접 지원 안 함 → 클라이언트 정렬)
    query = query.order('score_after', { ascending: false })
  } else if (sort === 'rating') {
    query = query.order('rating', { ascending: false })
  }

  const from = (page - 1) * limit
  query = query.range(from, from + limit - 1)

  const { data, error, count } = await query
  if (error) throw error

  // score_gain 정렬 시 클라이언트 재정렬
  if (sort === 'score_gain' && data) {
    data.sort((a, b) => (b.score_after - b.score_before) - (a.score_after - a.score_before))
  }

  return { data, count }
}

export async function getReviewStats() {
  const { data } = await supabase
    .from('reviews')
    .select('rating')
    .eq('approved', true)

  if (!data || data.length === 0) return { avgRating: 0, totalCount: 0 }

  const avgRating = data.reduce((sum, r) => sum + r.rating, 0) / data.length
  return {
    avgRating: Math.round(avgRating * 10) / 10,
    totalCount: data.length
  }
}
```

```tsx
// src/pages/admin/AdminReviewPanel.tsx

import { approveReview, rejectReview, getAdminPendingReviews } from '@/services/reviewService'
import { toast } from 'react-hot-toast'

export function AdminReviewPanel({ adminId }: { adminId: string }) {
  const [reviews, setReviews] = useState([])
  const [rejectReason, setRejectReason] = useState('')

  useEffect(() => {
    getAdminPendingReviews().then(setReviews)
  }, [])

  const handleApprove = async (reviewId: string) => {
    try {
      await approveReview(reviewId, adminId)
      toast.success('승인 완료 (+5일 보상 지급)')
      setReviews(prev => prev.filter((r: any) => r.id !== reviewId))
    } catch (e: any) {
      toast.error(e.message)
    }
  }

  const handleReject = async (reviewId: string) => {
    if (!rejectReason) {
      toast.error('거절 사유를 입력해주세요.')
      return
    }
    try {
      await rejectReview(reviewId, adminId, rejectReason)
      toast.success('거절 처리 완료')
      setReviews(prev => prev.filter((r: any) => r.id !== reviewId))
      setRejectReason('')
    } catch (e: any) {
      toast.error(e.message)
    }
  }

  return (
    <div className="admin-review-panel">
      <h2>후기 승인 관리 ({reviews.length}건 대기)</h2>

      {reviews.map((review: any) => (
        <div key={review.id} className="admin-review-item">
          <div className="meta">
            <span>{review.nickname}</span>
            <span>{review.score_before} → {review.score_after}점</span>
            <span>별점: {'⭐'.repeat(review.rating)}</span>
            <span>{new Date(review.created_at).toLocaleDateString('ko-KR')}</span>
          </div>
          <p className="content">{review.content}</p>
          <div className="actions">
            <button
              className="btn-approve"
              onClick={() => handleApprove(review.id)}
            >
              ✅ 승인 (+5일 지급)
            </button>
            <input
              placeholder="거절 사유 입력"
              value={rejectReason}
              onChange={e => setRejectReason(e.target.value)}
            />
            <button
              className="btn-reject"
              onClick={() => handleReject(review.id)}
            >
              ❌ 거절
            </button>
          </div>
        </div>
      ))}
    </div>
  )
}
```

## 📊 7. KPI 설계

### 7-1. 추적 쿼리

```sql
-- 랜딩 → 진단 전환율 (GA 이벤트 기반, DB 보조)
SELECT 
  DATE(created_at) AS date,
  COUNT(*) AS new_diagnostics
FROM diagnostic_results
WHERE created_at >= NOW() - INTERVAL '30 days'
GROUP BY DATE(created_at);

-- 후기 작성률 (프로그램 14일+ 사용자 대비)
SELECT
  COUNT(DISTINCT u.id) AS eligible_users,
  COUNT(DISTINCT r.user_id) AS review_writers,
  ROUND(
    COUNT(DISTINCT r.user_id)::numeric / NULLIF(COUNT(DISTINCT u.id), 0) * 100, 2
  ) AS write_rate_pct
FROM users u
LEFT JOIN reviews r ON r.user_id = u.id AND r.approved = TRUE
WHERE u.program_start_date <= NOW() - INTERVAL '14 days'
  AND u.program_status IN ('active', 'completed');

-- 후기 보상 지급 현황
SELECT
  COUNT(*) AS total_approved,
  COUNT(*) FILTER (WHERE rewarded = TRUE) AS rewarded_count,
  SUM(5) FILTER (WHERE rewarded = TRUE) AS total_bonus_days_given
FROM reviews
WHERE approved = TRUE;

-- 별점 분포
SELECT
  rating,
  COUNT(*) AS count,
  ROUND(COUNT(*)::numeric / SUM(COUNT(*)) OVER () * 100, 1) AS pct
FROM reviews
WHERE approved = TRUE
GROUP BY rating
ORDER BY rating DESC;
```
